## 9. Implementation Details

### Model: PasswordResetToken

```python
class PasswordResetToken(models.Model):
    """
    DB-backed password reset token (C1 fix).
    Replaces in-memory _PASSWORD_RESET_TOKENS dict.
    
    Lifecycle:
    1. generate_and_save(email) → creates token_hash, returns (obj, raw_token)
    2. Email sent with raw_token in reset URL
    3. verify_and_mark_used(raw_token) → validates, marks used, returns obj
    4. cleanup_expired() → removes expired unused tokens (periodic task)
    """
    
    email = EmailField(db_index=True)
    token_hash = CharField(unique=True, db_index=True)  # HMAC-SHA256
    created_at = DateTimeField(auto_now_add=True)
    expires_at = DateTimeField()  # default: now + 1 hour
    used = BooleanField(default=False, db_index=True)
    used_at = DateTimeField(null=True, blank=True)
    
    @property
    def is_expired(self) -> bool:
        return timezone.now() > self.expires_at
    
    @property
    def is_valid(self) -> bool:
        return not self.used and not self.is_expired
    
    @classmethod
    def generate_and_save(cls, email: str, hours: int = 1):
        """Generate token, hash, save to DB. Returns (token_obj, raw_token)."""
        raw_token = secrets.token_urlsafe(32)
        token_hash = _hash_token(raw_token)
        token_obj = cls.objects.create(
            email=email,
            token_hash=token_hash,
            expires_at=timezone.now() + timedelta(hours=hours),
        )
        return token_obj, raw_token
    
    @classmethod
    def verify_and_mark_used(cls, raw_token: str):
        """Hash token, look up, validate, mark used. Raises ValueError if invalid."""
        token_hash = _hash_token(raw_token)
        try:
            token = cls.objects.get(token_hash=token_hash)
        except cls.DoesNotExist:
            raise ValueError('Token is invalid')
        
        if token.used:
            raise ValueError('Token has already been used')
        if token.is_expired:
            raise ValueError('Token has expired')
        
        token.used = True
        token.used_at = timezone.now()
        token.save(update_fields=['used', 'used_at'])
        return token
    
    @classmethod
    def cleanup_expired(cls) -> int:
        """Delete expired tokens. Returns count deleted."""
        expired_count, _ = cls.objects.filter(
            expires_at__lt=timezone.now(),
            used=False
        ).delete()
        return expired_count
```

### Helper Function: _hash_token

```python
def _hash_token(raw_token: str) -> str:
    """HMAC-SHA256 hash with SECRET_KEY pepper."""
    import hashlib
    from django.conf import settings
    
    pepper = settings.SECRET_KEY.encode()
    token_bytes = raw_token.encode()
    return hashlib.sha256(token_bytes + pepper).hexdigest()
```
