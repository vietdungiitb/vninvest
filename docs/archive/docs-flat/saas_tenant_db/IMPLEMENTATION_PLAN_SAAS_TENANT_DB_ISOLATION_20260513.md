## 3. P0: Fix Migration Version Drift (backend/portfolio)

### 3.1 Root Cause phân tích

```
0041_tickersessionclose.py  — Generated by Django 5.2.3
  Creates TickerSessionClose với index names:
    portfolio_ti_market__cf6ff0_idx  ← Django 5.x auto-generated hash
    portfolio_ti_symbol__f849f1_idx  ← Django 5.x auto-generated hash

0042_remove_advisoryaggregatesnapshot_...py — Generated by Django 4.2.13
  RenameIndex: portfolio_ti_market__cf6ff0_idx → portfolio_t_market__2ec19a_idx
  RenameIndex: portfolio_ti_symbol__f849f1_idx → portfolio_t_symbol__f3e3e8_idx

CONFLICT: Django 4.2.x tạo index names khác với Django 5.x
  → Khi apply 0041 bằng Django 4.x, tên index sẽ là *khác* với tên trong 0041
  → 0042's RenameIndex không tìm được target → FAIL trên clean install
```

```python
def backfill_transaction_tenant(apps, schema_editor):
    """
    Gán tenant cho Transaction từ portfolio.tenant.
    Nếu portfolio.tenant is None, skip (sẽ xử lý sau khi portfolio backfill done).
    """
    Transaction = apps.get_model('user_workspace', 'Transaction')
    transactions = Transaction.objects.filter(
        tenant__isnull=True
    ).select_related('portfolio')
    
    updated = []
    for tx in transactions.iterator(chunk_size=500):
        if tx.portfolio and tx.portfolio.tenant_id:
            tx.tenant_id = tx.portfolio.tenant_id
            updated.append(tx)
        if len(updated) >= 500:
            Transaction.objects.bulk_update(updated, ['tenant_id'])
            updated = []
    if updated:
        Transaction.objects.bulk_update(updated, ['tenant_id'])

def backfill_subaccount_tenant(apps, schema_editor):
    """
    Gán tenant cho SubAccount từ portfolio.tenant hoặc user's primary tenant.
    """
    SubAccount = apps.get_model('user_workspace', 'SubAccount')
    subaccounts = SubAccount.objects.filter(
        tenant__isnull=True
    ).select_related('portfolio', 'user')
    
    updated = []
    for sa in subaccounts.iterator(chunk_size=500):
        if sa.portfolio and sa.portfolio.tenant_id:
            sa.tenant_id = sa.portfolio.tenant_id
            updated.append(sa)
        if len(updated) >= 500:
            SubAccount.objects.bulk_update(updated, ['tenant_id'])
            updated = []
    if updated:
        SubAccount.objects.bulk_update(updated, ['tenant_id'])
```

```python
# permissions.py (CREATE)
class TenantScopePermission(BasePermission):
    """
    Đảm bảo request.user có active tenant.
    Gắn request.tenant cho views sử dụng.
    """
    def has_permission(self, request, view) -> bool:
        if not request.user or not request.user.is_authenticated:
            return False
        tenant = _resolve_tenant(request.user)
        if tenant is None:
            return False
        request.tenant = tenant
        return True

def _resolve_tenant(user) -> Optional['CoreTenant']:
    """
    Tìm primary tenant của user qua UserProfile hoặc TenantMember.
    Return None nếu không có tenant (backward compat với single-tenant mode).
    """
    profile = getattr(user, 'core_profile', None)
    if profile and profile.tenant_id:
        return profile.tenant
    # Fallback: first TenantMember
    from identity_access.models import TenantMember
    member = TenantMember.objects.filter(user=user, is_active=True).select_related('tenant').first()
    return member.tenant if member else None
```

```python
def _get_portfolio_or_404(user, tenant, portfolio_id):
    """
    Lấy portfolio, double-check user VÀ tenant để block cross-tenant access.

    try:
```

**Parity check query:**

```python
# scripts/check_ingest_parity.py
from datetime import date, timedelta
from market_data.models import DailyPrice as CorePrice

def check_parity(check_date: date) -> dict:
    """
    So sánh DailyPrice trong core_db vs legacy vninvest DB.
    """
    core_count = CorePrice.objects.filter(date=check_date).count()
    
    # Legacy count via raw SQL trên legacy_readonly
    from django.db import connections
    with connections['legacy_readonly'].cursor() as cursor:
        cursor.execute(
            "SELECT COUNT(*) FROM portfolio_tickersessionclose WHERE session_date = %s",
            [check_date]
        )
        legacy_count = cursor.fetchone()[0]
    
    parity = min(core_count, legacy_count) / max(core_count, legacy_count, 1)
    return {
        'date': check_date.isoformat(),
        'core_count': core_count,
        'legacy_count': legacy_count,
        'parity_score': round(parity, 4),
        'pass': parity >= 0.99,
    }
```
